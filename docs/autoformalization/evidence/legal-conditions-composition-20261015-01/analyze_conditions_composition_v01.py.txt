def require(value,message):

def canonical(value):return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True,allow_nan=False).encode()


def digest(value):return hashlib.sha256(canonical(value)).hexdigest()


def f32(value):
    require(type(value) in (int,float) and type(value) is not bool and math.isfinite(value),'Finite scalar required')
    try:out=struct.unpack('<f',struct.pack('<f',value))[0]
    except (OverflowError,struct.error):raise ValueError('Float32 overflow') from None
    require(math.isfinite(out),'Finite float32 result required');return out

def raw_pin(path,expected=None):
    p=Path(path);require(p.is_absolute() and not forbidden(p) and p.resolve(strict=True)==p,'Exact nonprivate noncheckpoint physical path required')
    for part in (*reversed(p.parents),p):
        meta=part.lstat();require(not stat.S_ISLNK(meta.st_mode) and (stat.S_ISREG(meta.st_mode) if part==p else stat.S_ISDIR(meta.st_mode)),'Ordinary physical path required')
    before=p.stat();limit=MAX_OUTPUT if p==OUTPUT else MAX_PACKET;require(0<before.st_size<=limit,'Input/output byte bound exceeded')
    h=hashlib.sha256();n=0
    with p.open('rb') as stream:
        for block in iter(lambda:stream.read(1048576),b''):h.update(block);n+=len(block);require(n<=limit,'Input/output grew above bound')
    after=p.stat();identity=lambda s:(s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
    require(identity(before)==identity(after) and n==before.st_size,'Input changed while hashing')
    actual=dict(path=str(p),bytes=n,sha256=h.hexdigest())
    wanted=expected if expected is not None else EXPECTED.get(str(p))
    if wanted is not None:require(actual['bytes']==wanted['bytes'] and actual['sha256']==wanted['sha256'] and ('path' not in wanted or wanted['path']==str(p)),'Exact raw input identity mismatch')
    require(str(p) not in OBSERVED or OBSERVED[str(p)]==actual,'Earlier consumed input drift');OBSERVED[str(p)]=actual
    return actual

def align_literal(source,tokens,text):
    require(type(text) is str and text,'Fixed literal span required');starts={t['start']:i for i,t in enumerate(tokens)};ends={t['end']:i for i,t in enumerate(tokens)};matches=[];cursor=0
    while True:
        p=source.find(text,cursor)
        if p<0:break
        if p in starts and p+len(text) in ends:matches.append((starts[p],ends[p+len(text)]))
        cursor=p+1
    require(len(matches)==1,'Fixed gold/short literal must be unique and exactly token aligned');return matches[0]


def gap(a,b):return dict(binary64=a-b,f32=f32(a-b))


def boundary_rank(logits,gold,indices,short):

    require(all(type(x) is float and math.isfinite(x) and f32(x)==x for x in logits),'Exact finite binary32 boundary logits')
    ranked=sorted(indices,key=lambda i:(-logits[i],i));wrong=[i for i in ranked if i!=gold];best=logits[wrong[0]] if wrong else None
    a=None if best is None else gap(logits[gold],best);b=gap(logits[gold],logits[short])
    return dict(gold_index=gold,domain_indices=indices,stable_rank=ranked.index(gold)+1,
        strictly_higher=sum(logits[i]>logits[gold] for i in indices),exact_ties_including_gold=sum(logits[i]==logits[gold] for i in indices),
        guard_ties_including_gold=sum(abs(f32(logits[i]-logits[gold]))<=1e-7 for i in indices),
        in_marginal_top8=gold in ranked[:8],logit=logits[gold],gold_minus_best_wrong_binary64=None if a is None else a['binary64'],
        gold_minus_best_wrong_f32=None if a is None else a['f32'],gold_minus_short_binary64=b['binary64'],gold_minus_short_f32=b['f32'])


def full_span_rank(starts,ends,gold,short):
    require(len(starts)==len(ends) and 1<=len(starts)<=128 and all(type(v) is float and math.isfinite(v) and f32(v)==v for v in starts+ends),'Finite complete binary32 pair vectors')
    n=len(starts);require(all(type(i) is int for i in (*gold,*short)) and 0<=gold[0]<=gold[1]<n and short[0]==gold[0] and gold[0]<=short[1]<n,'Exact gold/short upper-triangle coordinates')
    pairs=[(f32(starts[l]+ends[r]),l,r) for l in range(n) for r in range(l,n)];pairs.sort(key=lambda a:(-a[0],a[1],a[2]));value=f32(starts[gold[0]]+ends[gold[1]]);shortvalue=f32(starts[short[0]]+ends[short[1]])
    rank=next(i+1 for i,p in enumerate(pairs) if p[1:]==tuple(gold));wrong=next((p for p in pairs if p[1:]!=tuple(gold)),None);margin=None if wrong is None else gap(value,wrong[0]);shortgap=gap(value,shortvalue)
    boundary=f32(pairs[7][0]-pairs[8][0]) if len(pairs)>8 else None
    return dict(gold_start=gold[0],gold_end=gold[1],stable_rank=rank,domain_count=len(pairs),strictly_higher=sum(p[0]>value for p in pairs),
        exact_ties_including_gold=sum(p[0]==value for p in pairs),guard_ties_including_gold=sum(abs(f32(p[0]-value))<=1e-7 for p in pairs),
        in_fullspan_top8=rank<=8,in_fullspan_top16=rank<=16,pair_score_f32=value,gold_minus_best_wrong_binary64=None if margin is None else margin['binary64'],
        gold_minus_best_wrong_f32=None if margin is None else margin['f32'],gold_minus_short_binary64=shortgap['binary64'],gold_minus_short_f32=shortgap['f32'],
        boundary8_9_margin_f32=boundary,boundary8_9_tied_within_guard=boundary is not None and boundary<=1e-7,
        top8=[dict(start=l,end=r,score_f32=v) for v,l,r in pairs[:8]])


def selection(record,route):
    if route=='raw':
        field=record.get('span_diagnostics',{}).get('facets',{}).get('conditions')
        coords=None if field is None or field.get('token_start') is None else dict(start=field['token_start'],end=field['token_end_inclusive'])
    else:

        coords=None if span is None else dict(start=span['token_start'],end=span['token_end_inclusive'])
    return dict(status=record['status'],reason=record['reason'],conditions=coords,canonical_ir=record['canonical_ir'])


def assignment_coverage(row,scores,coords,joint):

        gold_assignment_nonoverlap=nonoverlap,full_gold_assignment_retained=bool(fixed and all(coverage.values()) and nonoverlap),gold_dependent_postfreeze_diagnostic=True)
