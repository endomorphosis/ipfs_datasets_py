def require(value,message):
    if not value:raise ValueError(message)

def canonical(value):return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True,allow_nan=False).encode()

def digest(value):return hashlib.sha256(canonical(value)).hexdigest()

def f32(value):
    require(type(value) in (int,float) and type(value) is not bool and math.isfinite(value),'Finite scalar required')
    try:out=struct.unpack('<f',struct.pack('<f',value))[0]
    except (OverflowError,struct.error):raise ValueError('Float32 overflow') from None
    require(math.isfinite(out),'Finite float32 result required');return out

def descriptor(value):
    require(type(value) is dict and set(value)=={'path','bytes','sha256'} and type(value['path']) is str
        and Path(value['path']).is_absolute() and '..' not in Path(value['path']).parts
        and type(value['bytes']) is int and 0<value['bytes']<=MAX_PACKET
        and type(value['sha256']) is str and re.fullmatch('[0-9a-f]{64}',value['sha256']),'Closed finite raw descriptor required')
    return value

def forbidden(path):
    p=Path(path)
    return (any(part.startswith('private-') or part in {'.git','.cache','__pycache__'} for part in p.parts)
        or 'checkpoint' in p.name or p.suffix in ('.log','.jsonl')
        or p.name in {'resources-start.json','resources-final.json','scheduler.json','disk-reservations.json'}
        or p.name.startswith('ipfs-datasets-resource-scheduler-'))

def raw_pin(path,expected=None):
    p=Path(path);require(p.is_absolute() and not forbidden(p) and p.resolve(strict=True)==p,'Exact nonprivate noncheckpoint physical path required')
    for part in (*reversed(p.parents),p):
        meta=part.lstat();require(not stat.S_ISLNK(meta.st_mode) and (stat.S_ISREG(meta.st_mode) if part==p else stat.S_ISDIR(meta.st_mode)),'Ordinary physical path required')
    before=p.stat();limit=MAX_OUTPUT if p==OUTPUT else MAX_PACKET;require(0<before.st_size<=limit,'Input/output byte bound exceeded')
    h=hashlib.sha256();n=0
    with p.open('rb') as stream:
        for block in iter(lambda:stream.read(1048576),b''):h.update(block);n+=len(block);require(n<=limit,'Input/output grew above bound')
    after=p.stat();identity=lambda s:(s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
    require(identity(before)==identity(after) and n==before.st_size,'Input changed while hashing')
    actual=dict(path=str(p),bytes=n,sha256=h.hexdigest())
    wanted=expected if expected is not None else EXPECTED.get(str(p))
    if wanted is not None:require(actual['bytes']==wanted['bytes'] and actual['sha256']==wanted['sha256'] and ('path' not in wanted or wanted['path']==str(p)),'Exact raw input identity mismatch')
    require(str(p) not in OBSERVED or OBSERVED[str(p)]==actual,'Earlier consumed input drift');OBSERVED[str(p)]=actual
    return actual

def read_json(path,expected=None):
    p=Path(path);role=ROLES.get(str(p))
    require(role in {'metadata','reference','capture'} and (role=='metadata' or BODY_READY),'Reference/capture body read before complete donor custody prohibited')
    before=raw_pin(p,expected);raw=p.read_bytes();require(len(raw)==before['bytes'] and hashlib.sha256(raw).hexdigest()==before['sha256'],'Consumed JSON differs')
    def unique(pairs):
        out={}
        for k,v in pairs:require(k not in out,'Duplicate JSON key');out[k]=v
        return out
    def number(text):
        value=float(text);require(math.isfinite(value),'Nonfinite JSON number');return value
    value=json.loads(raw,object_pairs_hook=unique,parse_float=number,parse_constant=lambda _:(_ for _ in ()).throw(ValueError('Nonfinite JSON')))
    raw_pin(p,before);return value

class FiniteIO:
    def __init__(self):self.allowed={str(SELF),str(GATE),str(INPUTS)}
    def admit(self,artifacts,roles):
        require(type(artifacts) is dict and 0<len(artifacts)<=1024 and type(roles) is dict and set(roles)<=set(artifacts),'Finite reviewed artifact roles required')
        require(sum(v['bytes'] for v in artifacts.values())<=1000000000,'Total pinned closure above1GB')
        require(sum(role=='source' for role in roles.values())<=128,'Source closure above128')
        for path,role in roles.items():
            require(role in {'metadata','reference','capture','source'} and not forbidden(path),'Closed nonprivate role required')
            item=dict(path=path,**{k:artifacts[path][k] for k in ('bytes','sha256')});descriptor(item)
            require(Path(path).suffix==('.py' if role=='source' else '.json'),'Role/suffix mismatch')
            self.allowed.add(path);EXPECTED[path]=item;ROLES[path]=role
    def audit(self,event,args):
        if event=='import':require(args[0].split('.')[0] in sys.stdlib_module_names and args[0].split('.')[0] not in {'site','ctypes','subprocess','socket'},'Only prequalified stdlib, no model/numerical/CID/SDK imports')
        require(not event.startswith(('subprocess.','socket.','os.exec','os.spawn')) and event not in {'os.system','os.fork','os.posix_spawn','fcntl.flock','os.kill','os.killpg','os.mkdir','os.unlink','os.remove','os.rmdir','os.rename','os.replace','marshal.loads','os.listdir','os.scandir'},'Process/network/resource/directory/shared mutation prohibited')
        if event=='open':
            require(isinstance(args[0],(str,bytes)),'Anonymous FD body access refused')
            path=str(Path(os.fsdecode(args[0])).absolute());mode,flags=args[1:3]
            require(not forbidden(path),'Private/checkpoint/log body refused')
            writing=isinstance(mode,str) and any(c in mode for c in 'wax+') or isinstance(flags,int) and bool(flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND))
            if writing:require(path==str(OUTPUT) and isinstance(flags,int) and flags&os.O_EXCL,'Exclusive own diagnostic output only')
            else:require(path in self.allowed,'Unlisted body/source read refused')
        if event in {'os.chmod','os.chown','os.truncate','os.utime','os.link','os.symlink'}:require(event=='os.chmod' and str(Path(args[0]).absolute())==str(OUTPUT) and args[1]==0o444,'Only own final output sealing allowed')


def wall_alarm(signum, frame):
    raise TimeoutError('Independent recount hard wall bound exceeded')


def enforce_limits():
    for kind, cap in ((resource.RLIMIT_AS, 1610612736), (resource.RLIMIT_CPU, 600),
                      (resource.RLIMIT_FSIZE, MAX_OUTPUT)):
        soft, hard = resource.getrlimit(kind)
        target = cap if hard == resource.RLIM_INFINITY else min(cap, hard)
        resource.setrlimit(kind, (target, target))
    signal.signal(signal.SIGALRM, wall_alarm)
    signal.alarm(900)


def bounded_time():
    require(time.monotonic() - STARTED <= 900, 'Independent recount wall bound exceeded')


def load_shared_plumbing(item):
    """Compile only an explicit custody/reference roster; never primary main."""
    require(item['path'] == str(PRIMARY_SOURCE) and ROLES.get(item['path']) == 'source',
            'Exact source-qualified primary plumbing source required')
    raw_pin(item['path'], item)
    tree = ast.parse(PRIMARY_SOURCE.read_bytes())
    constant_names = {'DESIGN', 'EXECUTION_CONTRACT', 'EXECUTION_PIN', 'DESIGN_PIN',
                      'SOURCE_PINS', 'F15_CHECKPOINT', 'F15_STATE', 'SCOPES', 'COUNTS', 'FIXED_IDS'}
    function_names = {'exact_admission', 'source_nodes', 'load_kernels', 'qualified_donor', 'validate_native'}
    nodes = []
    seen_constants, seen_functions = set(), set()
    for node in tree.body:
        if isinstance(node, ast.Assign):
            chosen = {target.id for target in node.targets if isinstance(target, ast.Name)} & constant_names
            if chosen:
                require(len(node.targets) == 1 and len(chosen) == 1, 'Closed shared constant assignment')
                nodes.append(node); seen_constants.update(chosen)
        elif isinstance(node, ast.FunctionDef) and node.name in function_names:
            require(not node.decorator_list and not any(isinstance(child, (ast.Import, ast.ImportFrom))
                                                       for child in ast.walk(node)), 'Pure shared plumbing AST required')
            nodes.append(node); seen_functions.add(node.name)
    require(seen_constants == constant_names and seen_functions == function_names,
            'Complete explicit shared plumbing AST roster required')
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(PRIMARY_SOURCE) + '#custody-reference-only', 'exec'), globals())
    raw_pin(item['path'], item)
    return {'constant_names': sorted(constant_names), 'function_names': sorted(function_names),
            'primary_scientific_helpers_called': [], 'primary_main_executed': False}


def load_pure_kernel(item):
    require(ROLES.get(item['path']) == 'source', 'Independent kernel source role required')
    raw_pin(item['path'], item)
    raw = Path(item['path']).read_bytes()
    module = {'__name__': 'f19_independent_kernel', '__file__': item['path']}
    exec(compile(raw, item['path'], 'exec'), module)
    raw_pin(item['path'], item)
    return module


def native_source_identity(rows, raw, chosen, copy_kernel, joint):
    require(len(rows) == len(raw['occurrences']) == len(raw['inference']['rows']) == len(chosen),
            'Independent native source coverage differs')
    for row, occurrence, owner, selected in zip(rows, raw['occurrences'], raw['inference']['rows'], chosen, strict=True):
        bounded_time()
        source_hash = hashlib.sha256(row['source_text'].encode('utf-8')).hexdigest()
        require(row['id'] == occurrence['id'] == selected['id']
                and source_hash == occurrence['source_sha256'] == owner['source_sha256']
                == selected['result']['source_sha256'], 'Independent exact native source join differs')
        copy_kernel['validate_packet'](occurrence['scores'])
        clean = joint['_validated'](row['source_text'], occurrence['scores'])
        require(len(clean['tokens']) <= 128 and clean == occurrence['scores'],
                'Independent fixed 128-token complete head geometry differs')


def selected_conditions(record, route):
    if route == 'raw':
        facet = (record.get('span_diagnostics') or {}).get('facets', {}).get('conditions')
        position = None if facet is None or facet.get('token_start') is None else {
            'start': facet['token_start'], 'end': facet['token_end_inclusive']}
    else:
        facet = (record.get('span_diagnostics') or {}).get('conditions')
        span = None if facet is None else facet.get('span')
        position = None if span is None else {'start': span['token_start'], 'end': span['token_end_inclusive']}
    return {'status': record['status'], 'reason': record['reason'], 'conditions': position,
            'canonical_ir': record['canonical_ir']}


def assignment_packet(row, scores, rank_kernel):
    rule = row['canonical_ir']['rules'][0]
    tokens = scores['tokens']
    gold = {}
    for field in FIELDS:
        atom = rule[field][0] if field in FIELDS[3:] and rule[field] else '' if field in FIELDS[3:] else rule[field]
        if atom:
            aligned = rank_kernel['aligned_span'](row['source_text'], tokens, atom)
            gold[field] = (aligned['token_start'], aligned['token_end_inclusive'])
    modalities = scores['modality_logits']
    winner = min(range(3), key=lambda index: (-modalities[index], index))
    others = [value for index, value in enumerate(modalities) if index != winner]
    fixed = f32(modalities[winner] - max(others)) > 1e-7 and ('O', 'P', 'F')[winner] == rule['modality']
    presence = {}
    for field in OPTIONAL:
        zero, one = scores['presence_logits'][field]
        presence[field] = abs(f32(one - zero)) > 1e-7 and (one > zero) == (field in gold)
        fixed = bool(fixed and presence[field])
    covered = {}
    for field, (left, right) in gold.items():
        covered[field] = rank_kernel['span_rank'](scores['start_logits'][field], scores['end_logits'][field],
                                                 left, right)['retained_by_stable_top8']
    occupied = set(); nonoverlap = True
    for left, right in gold.values():
        for position in range(left, right + 1):
            if position in occupied: nonoverlap = False
            occupied.add(position)
    return {'fixed_modality_and_all_presence_correct': fixed, 'presence_correct': presence,
            'gold_present_role_fullspan_top8': covered, 'gold_assignment_nonoverlap': nonoverlap,
            'full_gold_assignment_retained': bool(fixed and all(covered.values()) and nonoverlap),
            'gold_dependent_postfreeze_diagnostic': True}


def independent_rank_row(row, occurrence, raw, chosen, donor, short_text, rank_kernel):
    scores = occurrence['scores']; tokens = scores['tokens']
    require(1 <= len(tokens) <= 128, 'Fixed diagnostic token budget required')
    truth = row['canonical_ir']['rules'][0]['conditions']
    require(type(truth) is list and len(truth) == 1, 'One fixed training conditions atom required')
    result = rank_kernel['primary_rank_fields'](row['source_text'], tokens, truth[0], short_text,
                                               scores['start_logits']['conditions'], scores['end_logits']['conditions'])
    result.update(id=row['id'], donor=donor,
                  source_sha256=hashlib.sha256(row['source_text'].encode('utf-8')).hexdigest(),
                  tokens_sha256=digest(tokens), raw_selection=selected_conditions(raw, 'raw'),
                  joint_selection=selected_conditions(chosen, 'joint'),
                  assignment_coverage=assignment_packet(row, scores, rank_kernel))
    return result


def panel_index(values):
    require(type(values) is list, 'Panel list required')
    indexed = {}
    for panel in values:
        require(type(panel) is dict and (panel['label'], panel['scope']) not in indexed,
                'Unique explicit panel label/scope required')
        indexed[panel['label'], panel['scope']] = panel
    return indexed
