def need(condition, message):
    if not condition:
        raise RuntimeError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def wire(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'),
        ensure_ascii=False, allow_nan=False).encode('utf-8')


def unique_pairs(pairs):
    result = {}
    for key, value in pairs:
        need(key not in result, 'duplicate metadata key refused')
        result[key] = value
    return result


def decode(raw):
    return json.loads(raw, object_pairs_hook=unique_pairs,
        parse_constant=lambda _: (_ for _ in ()).throw(ValueError('nonfinite metadata refused')))


def local_snapshot(path, maximum):
    """Bootstrap checked source bytes before loading the pinned native reader."""
    path = Path(path)
    need(path.is_absolute(), 'absolute bounded input required')
    for parent in (path.parent, *path.parent.parents):
        need(stat.S_ISDIR(parent.lstat().st_mode), 'unaliased input parents required')
    with os.fdopen(os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK), 'rb') as stream:
        before = os.fstat(stream.fileno())
        need(stat.S_ISREG(before.st_mode) and before.st_nlink == 1
            and 0 < before.st_size <= maximum, 'bounded single-link regular input required')
        raw = stream.read(maximum + 1)
        after = os.fstat(stream.fileno())
    identity = lambda info: (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns, info.st_ctime_ns)
    need(len(raw) == before.st_size and identity(before) == identity(after)
        == identity(path.lstat()), 'bounded input changed during read')
    return raw


def load_checked(path, pins, role):
    """Execute the exact checked buffer with its real __file__; never read pyc."""
    path = Path(path)
    need(str(path) in pins, 'helper source must be in frozen native request')
    raw = local_snapshot(path, MAX_SOURCE)
    need(sha(raw) == pins[str(path)], 'helper source SHA differs')
    name = '_qualified_definition_review_' + role + '_' + uuid.uuid4().hex
    need(name not in sys.modules, 'fresh helper module required')
    module = types.ModuleType(name)
    module.__file__ = str(path)
    module.__package__ = ''
    sys.modules[name] = module
    try:
        exec(compile(raw, str(path), 'exec', dont_inherit=True, optimize=0), module.__dict__)
    except BaseException:
        sys.modules.pop(name, None)
        raise
    return module


def code_guard(pins, snapshot):
    for spelling, digest in pins.items():
        need(sha(snapshot(Path(spelling), MAX_SOURCE)) == digest, 'frozen execution source changed')


def product_origins(root, pins):
    """Require every loaded repository product Python owner to be admitted."""
    origins = {}
    for name, module in tuple(sys.modules.items()):
        if name != 'ipfs_datasets_py' and not name.startswith('ipfs_datasets_py.'):
            continue
        file = getattr(module, '__file__', None)
        if file is None:
            continue
        path = Path(file)
        need(path.is_absolute() and path.suffix == '.py' and root in path.parents,
            'product owner must originate in the selected source checkout')
        need(str(path) in pins, 'loaded product owner missing from frozen source pins')
        origins[name] = {'path': str(path), 'sha256': pins[str(path)]}
    return origins

class PinnedInput:
    """Only read_bytes is adapted; the retained probe code stays unchanged."""
    def __init__(self, path, length, digest, snapshot):
        self.path, self.length, self.digest, self.snapshot = Path(path), length, digest, snapshot
        self.reads = 0

    def read_bytes(self):
        raw = self.snapshot(self.path, self.length)
        need(len(raw) == self.length and sha(raw) == self.digest,
            'retained original input differs from fixed code commitment')
        self.reads += 1
        return raw

    def descriptor(self):
        return {'path': str(self.path), 'bytes': self.length, 'sha256': self.digest}


def save(path, raw):
    path = Path(path)
    with os.fdopen(os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600), 'wb') as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    descriptor = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        os.fsync(descriptor)
    finally:
        os.close(descriptor)
    return {'path': str(path), 'bytes': len(raw), 'sha256': sha(raw)}


def read_descriptor(descriptor, snapshot, bound=MAX_RECORD):
    need(type(descriptor) is dict and set(descriptor) == {'path', 'bytes', 'sha256'},
        'exact replay file descriptor required')
    need(type(descriptor['bytes']) is int and 0 < descriptor['bytes'] <= bound,
        'bounded replay descriptor required')
    raw = snapshot(Path(descriptor['path']), bound)
    need(len(raw) == descriptor['bytes'] and sha(raw) == descriptor['sha256'],
        'persisted replay file differs')
    return raw


def pin_existing(path, snapshot, bound=MAX_RECORD):
    raw = snapshot(path, bound)
    return {'path': str(path), 'bytes': len(raw), 'sha256': sha(raw)}


def pin_probe_outputs(output, snapshot):
    """The retained run() has a fixed 18-file output roster; no tree scan."""
    paths = [(output / name, MAX_SAMPLING if name in
        ('sampling-report.json', 'historical-sampling-report.json') else MAX_RECORD)
        for name in ('source.html', 'acquisition-descriptor.json',
            'historical-sampling-report.json', 'sampling-report.json')]
    for family in ('fol', 'tdfol'):
        paths.extend((output / family / name, MAX_RECORD)
            for name in ('request.json', 'report.json', 'companion.json'))
        paths.extend((output / family / 'graph-bundle' / name,
            8 * 1024 * 1024 if name.endswith('.parquet') else MAX_RECORD)
            for name in ('nodes.parquet', 'edges.parquet', 'manifest.json', 'snapshot.json'))
    need(len(paths) == 18, 'fixed retained probe output roster differs')
    return [pin_existing(path, snapshot, bound) for path, bound in paths]

        'descriptor_sha256': sha(inputs['descriptor_raw']),

def check_open_packet(packet, context_owner):
    need(len(packet['dependency_slots']) == 6
        and [row['kind'] for row in packet['dependency_slots']] == list(context_owner.DEPENDENCY_KINDS),
        'all six fixed dependency kinds required')
    for row in packet['dependency_slots']:
        need(row['open'] is True and row['status'] == 'open_missing'
            and row['availability'] == 'missing' and row['attachments'] == []
            and row['semantic_attachment_verified'] is False
            and row['locator'] is row['expected_view_id'] is row['candidate_selector'] is None,
            'source-only context obligations must remain missing and unattached')
    need(all(value is False for value in packet['authority'].values()), 'context authority must remain false')
    for key in ('dependency_inventory_exhaustive', 'context_closure_verified', 'source_semantics_verified',
            'independently_reviewed', 'training_qualified', 'proof_authority',
            'semantic_attachments_inferred', 'raw_html_coordinate_mapping_verified'):
        need(packet[key] is False, 'context qualification must remain false')

def versions():

    for name in ('pytest', 'cryptography', 'pyarrow', 'multiformats'):
        try:

    return {'installed_distribution_metadata': values,
        'external_distribution_integrity_or_trust_attested': False}
